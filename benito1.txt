if ($_SERVER['REQUEST_METHOD'] == 'POST' && isset($_POST['action']) && $_POST['action'] == 'checkin') {
  $member_id = $_POST['member_id'] ?? '';
  $check_in_time = date('Y-m-d H:i:s');
  if (!empty($member_id)) {
    $stmt = $mysqli->prepare("INSERT INTO attendance (member_id, check_in_time) VALUES (?, ?)");
    if ($stmt) {
      $stmt->bind_param("is", $member_id, $check_in_time);
      if ($stmt->execute()) {
        $message = "Check-in recorded successfully!";
        $message_type = "success";
      } else {
        $message = "Error: " . $stmt->error;
        $message_type = "error";
      }
      $stmt->close();
    } else {
      $message = "DB error: " . $mysqli->error;
      $message_type = "error";
    }
  }
}

// Handle check-out
if ($_SERVER['REQUEST_METHOD'] == 'POST' && isset($_POST['action']) && $_POST['action'] == 'checkout') {
  $attendance_id = $_POST['attendance_id'] ?? '';
  $check_out_time = date('Y-m-d H:i:s');
  if (!empty($attendance_id)) {
    $stmt = $mysqli->prepare("UPDATE attendance SET check_out_time = ? WHERE id = ? AND check_out_time IS NULL");
    if ($stmt) {
      $stmt->bind_param("si", $check_out_time, $attendance_id);
      if ($stmt->execute()) {
        $message = "Check-out recorded successfully!";
        $message_type = "success";
      } else {
        $message = "Error: " . $stmt->error;
        $message_type = "error";
      }
      $stmt->close();
    } else {
      $message = "DB error: " . $mysqli->error;
      $message_type = "error";
    }
  }
}

// Handle delete attendance
if ($_SERVER['REQUEST_METHOD'] == 'POST' && isset($_POST['action']) && $_POST['action'] == 'delete') {
  $attendance_id = $_POST['attendance_id'] ?? '';
  if (!empty($attendance_id)) {
    $stmt = $mysqli->prepare("DELETE FROM attendance WHERE id = ?");
    if ($stmt) {
      $stmt->bind_param('i', $attendance_id);
      if ($stmt->execute()) {
        $message = 'Attendance record deleted.';
        $message_type = 'success';
      } else {
        $message = 'Error deleting record: ' . $stmt->error;
        $message_type = 'error';
      }
      $stmt->close();
    } else {
      $message = 'DB error: ' . $mysqli->error;
      $message_type = 'error';
    }
  }
}

    $stmt->bind_param('sss', $like, $like, $like);

-- Gym Management System Database Schema

CREATE DATABASE IF NOT EXISTS gym_management;
USE gym_management;

-- Membership Plans Table (must be created first due to foreign key)
CREATE TABLE IF NOT EXISTS membership_plans (
    id INT AUTO_INCREMENT PRIMARY KEY,
    plan_name VARCHAR(100) NOT NULL,
    duration_days INT NOT NULL,
    price DECIMAL(10, 2) NOT NULL,
    description TEXT,
    status ENUM('Active', 'Inactive') DEFAULT 'Active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Members Table
CREATE TABLE IF NOT EXISTS members (
    id INT AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(20) NOT NULL,
    address TEXT,
    date_of_birth DATE,
    gender ENUM('Male', 'Female', 'Other') NOT NULL,
    registration_date DATE NOT NULL,
    membership_plan_id INT,
    status ENUM('Active', 'Inactive', 'Expired') DEFAULT 'Active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (membership_plan_id) REFERENCES membership_plans(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Payments Table
CREATE TABLE IF NOT EXISTS payments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    member_id INT NOT NULL,
    membership_plan_id INT NOT NULL,
    amount DECIMAL(10, 2) NOT NULL,
    payment_date DATE NOT NULL,
    payment_method ENUM('Cash', 'Card', 'Online', 'Other') DEFAULT 'Cash',
    status ENUM('Paid', 'Pending', 'Failed') DEFAULT 'Paid',
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (member_id) REFERENCES members(id) ON DELETE CASCADE,
    FOREIGN KEY (membership_plan_id) REFERENCES membership_plans(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Attendance Table
CREATE TABLE IF NOT EXISTS attendance (
    id INT AUTO_INCREMENT PRIMARY KEY,
    member_id INT NOT NULL,
    check_in_time DATETIME NOT NULL,
    check_out_time DATETIME NULL,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (member_id) REFERENCES members(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

<?php
session_start();

// Mark that the visitor has seen the site in this browser session so
// subsequent navigation to Home won't force the login while the browser is open.
if (empty($_SESSION['seen_in_session'])) {
    $_SESSION['seen_in_session'] = true;
}

$login_error = '';
$register_error = '';
$register_success = '';

// Check if user is already logged in
if (isset($_SESSION['user_logged_in']) && $_SESSION['user_logged_in'] === true) {
    header("Location: index.php");
    exit();
}

// Handle form submissions (login or register)
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $action = isset($_POST['action']) ? $_POST['action'] : 'login';

    // path for storing registered users (simple JSON store)
    $users_file = __DIR__ . '/users.json';
    $users = [];
    if (file_exists($users_file)) {
        $raw = file_get_contents($users_file);
        $decoded = json_decode($raw, true);
        if (is_array($decoded)) {
            $users = $decoded;
        }
    }

    if ($action === 'register') {
        $email = isset($_POST['reg_email']) ? trim($_POST['reg_email']) : '';
        $password = isset($_POST['reg_password']) ? $_POST['reg_password'] : '';
        $confirm = isset($_POST['reg_confirm']) ? $_POST['reg_confirm'] : '';

        if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
            $register_error = 'Please provide a valid email address.';
        } elseif (strlen($password) < 4) {
            $register_error = 'Password must be at least 4 characters.';
        } elseif ($password !== $confirm) {
            $register_error = 'Password and confirm password do not match.';
        } else {
            // check duplicate
            $exists = false;
            foreach ($users as $u) {
                if (isset($u['email']) && strcasecmp($u['email'], $email) === 0) {
                    $exists = true;
                    break;
                }
            }
            if ($exists) {
                $register_error = 'An account with that email already exists.';
            } else {
                $hashed = password_hash($password, PASSWORD_DEFAULT);
                $users[] = [
                    'email' => $email,
                    'password' => $hashed,
                    'created_at' => date('c')
                ];
                // save
                file_put_contents($users_file, json_encode($users, JSON_PRETTY_PRINT), LOCK_EX);
                $register_success = 'Registration successful. You can now login.';
            }
        }
    } else { // login
        $email = isset($_POST['email']) ? trim($_POST['email']) : '';
        $password = isset($_POST['password']) ? $_POST['password'] : '';

        $authenticated = false;
        // check registered users first
        foreach ($users as $u) {
            if (isset($u['email']) && strcasecmp($u['email'], $email) === 0) {
                if (isset($u['password']) && password_verify($password, $u['password'])) {
                    $authenticated = true;
                }
                break;
            }
        }

        // fallback to demo hardcoded credentials
        if (!$authenticated) {
            $correct_email = 'benny@gmail.com';
            $correct_password = 'benny';
            if ($email === $correct_email && $password === $correct_password) {
                $authenticated = true;
            }
        }

        if ($authenticated) {
            $_SESSION['user_logged_in'] = true;
            $_SESSION['user_email'] = $email;
            header("Location: index.php");
            exit();
        } else {
            $login_error = 'Invalid email or password. Please try again.';
        }
    }
}

<?php

/* ===== DELETE MEMBER (SOFT DELETE) ===== */
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'delete_member') {
  $member_id = isset($_POST['member_id']) ? (int)$_POST['member_id'] : 0;
  $success = false;

  if ($member_id > 0) {
    $stmt = $mysqli->prepare("UPDATE members SET status = 'Deleted' WHERE id = ?");
    if ($stmt) {
      $stmt->bind_param('i', $member_id);
      $success = $stmt->execute();
      $stmt->close();
    }
  }

  $is_ajax = !empty($_SERVER['HTTP_X_REQUESTED_WITH']) &&
             strtolower($_SERVER['HTTP_X_REQUESTED_WITH']) === 'xmlhttprequest';

  if ($is_ajax) {
    header('Content-Type: application/json');
    echo json_encode(['success' => (bool)$success, 'member_id' => $member_id]);
    exit();
  }

  header('Location: members.php');
  exit();
}

<?php

$error = '';
$success = '';

$mysqli = db_connect();

// Ensure member_profiles table exists (create minimal schema if missing)
$create_profiles_sql = "CREATE TABLE IF NOT EXISTS member_profiles (
  id INT AUTO_INCREMENT PRIMARY KEY,
  member_id INT NOT NULL,
  fitness_goals TEXT,
  experience_level VARCHAR(50),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;";
try {
  $mysqli->query($create_profiles_sql);
} catch (Exception $e) {
  // ignore/create attempt failures - insertion will handle errors later
}

// Load membership plans for the select field
$plans = [];
$plan_res = $mysqli->query("SELECT id, plan_name FROM membership_plans ORDER BY id ASC");
if ($plan_res) {
    while ($row = $plan_res->fetch_assoc()) {
        $plans[] = $row;
    }
    $plan_res->free();
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // basic sanitization
    $first_name = trim($_POST['first_name'] ?? '');
    $last_name = trim($_POST['last_name'] ?? '');
    $email = trim($_POST['email'] ?? '');
    $phone = trim($_POST['phone'] ?? '');
    $address = trim($_POST['address'] ?? '');
    $dob = trim($_POST['date_of_birth'] ?? null);
    $gender = $_POST['gender'] ?? 'Other';
    $membership_plan_id = !empty($_POST['membership_plan_id']) ? (int)$_POST['membership_plan_id'] : null;
    $fitness_goals = trim($_POST['fitness_goals'] ?? '');
    $experience_level = $_POST['experience_level'] ?? '';

    if ($first_name === '' || $last_name === '' || $email === '' || $phone === '') {
        $error = 'Please fill in all required fields (first name, last name, email, phone).';
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $error = 'Please provide a valid email address.';
    } else {
        // insert into members
        $stmt = $mysqli->prepare("INSERT INTO members (first_name, last_name, email, phone, address, date_of_birth, gender, registration_date, membership_plan_id, status) VALUES (?, ?, ?, ?, ?, ?, ?, CURDATE(), ?, 'Active')");
        if ($stmt) {
            $stmt->bind_param('sssssssi', $first_name, $last_name, $email, $phone, $address, $dob, $gender, $membership_plan_id);
            if ($stmt->execute()) {
                $member_id = $stmt->insert_id;
                $stmt->close();

                // store extended profile info (without emergency contact fields)
                $p = $mysqli->prepare("INSERT INTO member_profiles (member_id, fitness_goals, experience_level) VALUES (?, ?, ?)");
                if ($p) {
                  $p->bind_param('iss', $member_id, $fitness_goals, $experience_level);
                  $p->execute();
                  $p->close();
                }

                // Prepare registered member details to display
                $registered_member = [
                  'id' => $member_id,
                  'first_name' => $first_name,
                  'last_name' => $last_name,
                  'phone' => $phone,
                  'membership_plan_id' => $membership_plan_id
                ];

                // fetch plan details for start/end calculation
                $plan_name = 'N/A';
                $start_date = date('Y-m-d');
                $end_date = null;
                if (!empty($membership_plan_id)) {
                  $q = $mysqli->prepare('SELECT plan_name, duration_days FROM membership_plans WHERE id = ? LIMIT 1');
                  if ($q) {
                    $q->bind_param('i', $membership_plan_id);
                    $q->execute();
                    $res = $q->get_result();
                    if ($row = $res->fetch_assoc()) {
                      $plan_name = $row['plan_name'];
                      $duration = (int)$row['duration_days'];
                      $dt = new DateTime($start_date);
                      $dt->modify('+' . $duration . ' days');
                      $end_date = $dt->format('Y-m-d');
                    }
                    $q->close();
                  }
                }

                $success = 'Registration successful.';
                // attach extra info for rendering below
                $registered_member['plan_name'] = $plan_name;
                $registered_member['start_date'] = $start_date;
                $registered_member['end_date'] = $end_date;
            } else {
                $error = 'Failed to register member: ' . $stmt->error;
                $stmt->close();
            }
        } else {
            $error = 'Database error: ' . $mysqli->error;
        }
    }
}
